#NoEnv
#SingleInstance Force
#Persistent
#InstallKeybdHook
SendMode, Input
SetBatchLines, -1
ListLines, Off

; =============================
; ALT+F MAKRO
; =============================

toggleAltF := false
counterF := 0

c::
KeyWait, c
toggleAltF := !toggleAltF

if (toggleAltF)
{
    counterF := 0
    Gosub, SpamKeysF
    SetTimer, SpamKeysF, 1290
}
else
{
    SetTimer, SpamKeysF, Off
}
return

SpamKeysF:
global toggleAltF, counterF
if (!toggleAltF)
    return
Critical
counterF++

if (counterF >= 4)
{
    SendInput, {1 down}{2 down}{7 down}
    Sleep, 15
    SendInput, {1 up}{2 up}{7 up}{SC029}
    counterF := 0
}
else
{
    SendInput, {1 down}{2 down}
    Sleep, 15
    SendInput, {1 up}{2 up}{SC029}
}
return


; =============================
; ALT+G MAKRO
; =============================

}

}

Up::
SendInput {4 down}{5 down}
Sleep 15
SendInput {4 up}{5 up}
Return
